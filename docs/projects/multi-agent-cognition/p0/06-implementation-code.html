import { EventEmitter } from 'events'; import { AgentRole, AgentState, Task, Checkpoint } from './types'; export class AgentNode extends EventEmitter { readonly id: string; readonly role: AgentRole; private _state: AgentState = AgentState.IDLE; private parent?: AgentNode; private children: AgentNode[] = []; private currentTask?: Task; constructor( id: string, role: AgentRole, private checkpointService: CheckpointService, private llmClient: LLMClient ) { super(); this.id = id; this.role = role; } // State machine transition with checkpoint private async transition(newState: AgentState): Promise<void> { const prevState = this._state; this._state = newState; // Persist before emitting — failure here means no state change await this.checkpointService.save({ agentId: this.id, taskId: this.currentTask?.id, state: newState, context: this.getContext(), timestamp: new Date(), }); this.emit('stateChange', { prevState, newState, agentId: this.id }); } // Execute task with full lifecycle management async executeTask(task: Task): Promise<TaskResult> { this.currentTask = task; try { await this.transition(AgentState.ASSIGNED); // Planning phase — decompose task await this.transition(AgentState.PLANNING); const plan = await this.createPlan(task); // Delegate subtasks if needed if (plan.subtasks.length > 0) { await this.transition(AgentState.DELEGATING); const subtaskResults = await this.delegateSubtasks(plan.subtasks); await this.transition(AgentState.SUPERVISING); await this.waitForChildren(subtaskResults); } // Execute own work await this.transition(AgentState.EXECUTING); const result = await this.doWork(task, plan); // Await parent review if not root if (this.parent) { await this.transition(AgentState.AWAITING_REVIEW); } await this.transition(AgentState.COMPLETE); return result; } catch (error) { await this.handleError(error); throw error; } finally { await this.transition(AgentState.IDLE); this.currentTask = undefined; } } // Stream LLM output while executing private async doWork(task: Task, plan: Plan): Promise<TaskResult> { const stream = this.llmClient.streamCompletion({ role: this.role, task: task.description, context: this.getContext(), }); let fullResponse = ''; for await (const chunk of stream) { fullResponse += chunk; this.emit('llmStream', { agentId: this.id, chunk }); } return { output: fullResponse, artifacts: [] }; } }

package eventbus import ( "context" "encoding/json" "github.com/jackc/pgx/v5/pgxpool" ) type Listener struct { pool *pgxpool.Pool router *TopicRouter tenantID string } // Start listening to PostgreSQL NOTIFY channels func (l *Listener) Start(ctx context.Context) error { conn, err := l.pool.Acquire(ctx) if err != nil { return err } defer conn.Release() // Subscribe to tenant-specific channel channel := "events_" + l.tenantID _, err = conn.Exec(ctx, "LISTEN "+channel) if err != nil { return err } for { select { case <-ctx.Done(): return ctx.Err() default: notif, err := conn.Conn().WaitForNotification(ctx) if err != nil { continue } go l.handleNotification(notif) } } } func (l *Listener) handleNotification(n *pgconn.Notification) { var event Event if err := json.Unmarshal([]byte(n.Payload), &event); err != nil { return // Log error in production } // Route to appropriate topic handlers l.router.Route(event) }
Go internal/eventbus/router.go
package eventbus import ( "strings" "sync" ) type TopicRouter struct { mu sync.RWMutex handlers map[string][]Handler } type Handler func(Event) // Subscribe to a topic pattern (supports wildcards) func (r *TopicRouter) Subscribe(pattern string, h Handler) { r.mu.Lock() defer r.mu.Unlock() r.handlers[pattern] = append(r.handlers[pattern], h) } // Route event to matching handlers func (r *TopicRouter) Route(e Event) { r.mu.RLock() defer r.mu.RUnlock() for pattern, handlers := range r.handlers { if matchTopic(pattern, e.Topic) { for _, h := range handlers { go h(e) // Async dispatch } } } } // Pattern matching: "agent.state.*" matches "agent.state.changed" func matchTopic(pattern, topic string) bool { if pattern == "*" { return true } if strings.HasSuffix(pattern, ".*") { prefix := strings.TrimSuffix(pattern, ".*") return strings.HasPrefix(topic, prefix) } return pattern == topic }

WebSocket Bridge — Client Protocol Handler

TypeScript src/ws/bridge.ts
import { WebSocket, WebSocketServer } from 'ws'; import { EventBus } from '../eventbus'; interface ClientConnection { ws: WebSocket; tenantId: string; subscriptions: Set<string>; // OCI IDs lastSeq: number; } export class WebSocketBridge { private connections = new Map<string, ClientConnection>(); private eventBuffer = new Map<string, Event[]>(); // Reconnection buffer constructor( private wss: WebSocketServer, private eventBus: EventBus, private coalescer: EventCoalescer ) { this.setupEventBusSubscription(); this.setupWebSocketHandlers(); } private setupEventBusSubscription() { // Subscribe to all agent/task events this.eventBus.subscribe('agent.*', (e) => this.handleInternalEvent(e)); this.eventBus.subscribe('task.*', (e) => this.handleInternalEvent(e)); this.eventBus.subscribe('org.*', (e) => this.handleInternalEvent(e)); } private handleInternalEvent(event: InternalEvent) { // Transform internal event to client protocol const wsMessage = this.transformToClientMessage(event); // Find all clients subscribed to this OCI for (const [connId, conn] of this.connections) { if (conn.subscriptions.has(event.ociId)) { // Coalesce high-frequency events (LLM tokens) if (event.type === 'llm.token') { this.coalescer.add(connId, wsMessage); } else { this.sendToClient(conn, wsMessage); } } } // Buffer for potential reconnections this.bufferEvent(event.ociId, wsMessage); } private handleClientMessage(conn: ClientConnection, data: string) { const msg = JSON.parse(data) as ClientMessage; switch (msg.type) { case 'SUBSCRIBE': msg.ociIds.forEach(id => conn.subscriptions.add(id)); break; case 'UNSUBSCRIBE': msg.ociIds.forEach(id => conn.subscriptions.delete(id)); break; case 'TASK_SUBMIT': this.eventBus.publish('task.submitted', { ociId: msg.ociId, task: msg.task, tenantId: conn.tenantId, }); break; case 'RECONNECT': this.handleReconnect(conn, msg.lastSeq); break; case 'PING': this.sendToClient(conn, { type: 'PONG', ts: Date.now() }); break; } } // Replay missed events on reconnection private handleReconnect(conn: ClientConnection, lastSeq: number) { for (const ociId of conn.subscriptions) { const buffer = this.eventBuffer.get(ociId) ?? []; const missed = buffer.filter(e => e.seq > lastSeq); missed.forEach(e => this.sendToClient(conn, e)); } } }

🎯 Implementation Patterns Summary

1. Event Sourcing Lite: Checkpoints capture state snapshots, not full event streams — balances durability with storage efficiency.

2. Hierarchical State Machines: Each agent independently manages its own state machine; parent agents orchestrate child lifecycle.

3. Protocol Translation: WebSocket Bridge decouples internal event format from client protocol — enables evolution without breaking clients.

4. Tenant-Scoped NOTIFY: PostgreSQL channels are tenant-isolated by naming convention, eliminating cross-tenant event leakage.
